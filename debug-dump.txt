import axios from "axios";
import { API_URL, ADMIN_BASE } from "../config/settings";
import { getToken, clearToken } from "./authStorage";

const http = axios.create({ baseURL: API_URL });

http.interceptors.request.use((config) => {
  const token = getToken();
  if (token) config.headers.Authorization = `Bearer ${token}`;
  return config;
});

http.interceptors.response.use(
  (res) => res,
  (err) => {
    const status = err.response?.status;
    const onLoginPage = window.location.pathname.endsWith("/login");
    const hadToken = Boolean(getToken());

    // ÙÙ‚Ø· Ø§Ú¯Ø± ÙˆØ§Ù‚Ø¹Ø§Ù‹ ØªÙˆÚ©Ù† Ø¯Ø§Ø´ØªÛŒÙ… Ùˆ 401 Ú¯Ø±ÙØªÛŒÙ… â†’ Ù„Ø§Ú¯â€ŒØ§ÙˆØª

// Token and roles storage
import { TOKEN_KEY } from "../config/settings";

const ROLES_KEY = "admin_roles";

export const getToken = () => localStorage.getItem(TOKEN_KEY);
export const getRoles = () =>
  JSON.parse(localStorage.getItem(ROLES_KEY) || "[]");

// Ø¨Ø¹Ø¯ Ø§Ø² Ù„Ø§Ú¯ÛŒÙ† Ù…ÙˆÙÙ‚ ØµØ¯Ø§ Ø²Ø¯Ù‡ Ù…ÛŒâ€ŒØ´ÙˆØ¯
export function saveLogin({ token, roles = [] }) {
  localStorage.setItem(TOKEN_KEY, token);
  localStorage.setItem(ROLES_KEY, JSON.stringify(roles));
}

export function clearToken() {
  localStorage.removeItem(TOKEN_KEY);
  localStorage.removeItem(ROLES_KEY);
}

// Ø²Ù…Ø§Ù† Ø§Ù†Ù‚Ø¶Ø§ Ø§Ø² Ø®ÙˆØ¯ ØªÙˆÚ©Ù† Ø®ÙˆØ§Ù†Ø¯Ù‡ Ù…ÛŒâ€ŒØ´ÙˆØ¯ (ÙÛŒÙ„Ø¯ exp Ø¨Ù‡ Ø«Ø§Ù†ÛŒÙ‡)
function isExpired(token) {
  try {
    const part = token.split(".")[1].replace(/-/g, "+").replace(/_/g, "/");
    return JSON.parse(atob(part)).exp * 1000 < Date.now();
  } catch {
    return true;
  }
}

export const isLoggedIn = () => {
  const token = getToken();
  return Boolean(token) && !isExpired(token);
};

const fillPath = (template, values = {}) =>
  template.replace(/\{([^}]+)\}/g, (match, key) => {
    const value = values[key] ?? values[key[0].toLowerCase() + key.slice(1)];
    if (value === undefined || value === null || value === "") {
      throw new Error(`Path parameter ${key} was not provided for this request.`);
    }
    return encodeURIComponent(String(value));
  });

export const requestAcademyApi = (endpointName, options = {}) => {
  const endpoint = academyEndpoints.find((item) => item.name === endpointName);
  if (!endpoint) throw new Error(`API is unknown: ${endpointName}`);
  const { path = {}, params, data, config = {} } = options;
  return http.request({
    ...config,
    method: endpoint.method,
    url: fillPath(endpoint.path, path),
    ...(params !== undefined ? { params } : {}),
    ...(data !== undefined ? { data } : {}),
  });
};

export const academyApi = Object.freeze(
  Object.fromEntries(
    academyEndpoints.map(({ name }) => [
      name,
      (options = {}) => requestAcademyApi(name, options),
    ]),
  ),
);

export const getAcademyEndpoint = (name) =>
  academyEndpoints.find((item) => item.name === name);

  const [loading, setLoading] = useState(true);
  const [error, setError] = useState("");
  const [text, setText] = useState("");
  const [sending, setSending] = useState(false);

    if (!id) return;

        setMessages(arr);

    if (!text.trim()) return;

    try {
      await sendSupportMessage({
        ticketId: id,
        message: text.trim(),
        text: text.trim(),
      });
      toast.success("Message sent");
      setText("");
      const res = await getTicketChat(id);
      const data = res.data?.data ?? res.data?.result ?? res.data;
      const arr = Array.isArray(data)
        ? data
        : (data?.messages ?? data?.chat ?? []);
      setMessages(arr);
      onChanged?.();
    } catch (err) {
      toast.error(errorText(err));
    } finally {
      setSending(false);
    }
  };

  const msgText = (m) => pick(m, ["message", "text", "content", "body"]) || "";
  const msgSender = (m) =>
    pick(m, ["sender", "senderName", "userName", "role"]) || "";
  const msgDate = (m) => pick(m, ["createdAt", "date", "insertDate"]) || "";

  return (
    <Modal title={`Ticket chat: ${tkTitle(ticket)}`} onClose={onClose}>

        </p>

        <>
          {error && <p className="mb-3 text-sm text-error">{error}</p>}
          <div className="mb-4 max-h-80 space-y-2 overflow-y-auto">
            {messages.length === 0 ? (
              <p className="py-8 text-center text-sm text-muted-foreground">
                No messages
              </p>
            ) : (
              messages.map((m, i) => (
                <div
                  key={i}
                  className="rounded-xl bg-surface-2 p-3 dark:bg-muted"
                >
                  <div className="mb-1 flex items-center justify-between text-[11px] text-muted-foreground">
                    <span className="font-medium">{msgSender(m)}</span>
                    <span dir="ltr">{msgDate(m)}</span>

                  <p className="text-body-sm">{msgText(m)}</p>

              ))
            )}

              onChange={(e) => setText(e.target.value)}
              onKeyDown={(e) => e.key === "Enter" && !e.shiftKey && send()}

// Convert the server response to { rows, total }
export function normalizeList(data) {
  if (Array.isArray(data)) return { rows: data, total: data.length };
  const box = data?.data ?? data?.result ?? data ?? {};
  if (Array.isArray(box))
    return { rows: box, total: data?.totalCount ?? box.length };
  const arrayKey =
    listKeys.find((k) => Array.isArray(box[k])) ??
    Object.keys(box).find((k) => Array.isArray(box[k]));
  const rows = arrayKey ? box[arrayKey] : [];
  const total = box.totalCount ?? box.total ?? box.totalRecords ?? rows.length;
  return { rows, total };
}

import { useEffect, useState } from "react";
import { normalizeList, errorText } from "../lib/apiData";

export function useListData(fetcher, params, refreshKey) {
  const [state, setState] = useState({
    rows: [],
    total: 0,
    loading: true,
    error: "",
  });
  const paramsKey = JSON.stringify(params);

  useEffect(() => {
    let cancelled = false; // If the component was closed, ignore the old response
    setState((old) => ({ ...old, loading: true, error: "" }));
    fetcher(params)
      .then(
        (res) =>
          !cancelled &&
          setState({ ...normalizeList(res.data), loading: false, error: "" }),
      )
      .catch(
        (err) =>
          !cancelled &&
          setState({
            rows: [],
            total: 0,
            loading: false,
            error: errorText(err),
          }),
      );
    return () => {
      cancelled = true;
    };
  }, [paramsKey, refreshKey]);

  return state;
}
